<script>
(() => {
  const canvas = document.getElementById('game');
  const ctx = canvas.getContext('2d');
  const scoreEl = document.getElementById('score');
  const healthEl = document.getElementById('health');
  const statusEl = document.getElementById('statusText');
  const intro = document.getElementById('intro');
  const gameover = document.getElementById('gameover');
  const finalScore = document.getElementById('finalScore');
  const bestScore = document.getElementById('bestScore');
  const overTitle = document.getElementById('overTitle');
  const overText = document.getElementById('overText');
  const overKicker = document.getElementById('overKicker');
  const toast = document.getElementById('toast');
  const themeBtn = document.getElementById('themeBtn');
  const state = {
    running:false, won:false, score:0, best:0, health:3, target:12,
    time:0, shake:0, spawnTimer:0, dashCd:0,
    keys:new Set(), particles:[], packets:[], glitches:[], stars:[]
  };

  const player = {x:0,y:0,r:14,vx:0,vy:0,speed:240,dash:0,trail:[]};
  const pointer = {x:0,y:0};

  function resize(){
    const dpr = Math.min(devicePixelRatio || 1, 2);
    canvas.width = innerWidth * dpr; canvas.height = innerHeight * dpr;
    canvas.style.width = innerWidth + 'px'; canvas.style.height = innerHeight + 'px';
    ctx.setTransform(dpr,0,0,dpr,0,0);
    if (!state.running && !state.won && state.score===0) centerPlayer();
    buildStars();
  }

  function centerPlayer(){ player.x = innerWidth/2; player.y = innerHeight/2; player.vx = 0; player.vy = 0; }

  function buildStars(){
    state.stars = Array.from({length:Math.max(80, Math.floor(innerWidth*innerHeight/12000))}, (_,i)=>({
      x:(i*97)%Math.max(innerWidth,1), y:(i*53)%Math.max(innerHeight,1),
      z:Math.random()*1+.2, s:Math.random()*2+0.5
    }));
  }

  function reset(){
    state.running = true; state.won = false; state.score = 0; state.health = 3; state.time = 0; state.shake = 0; state.spawnTimer = 0; state.dashCd = 0;
    state.particles = []; state.packets = []; state.glitches = []; player.trail = [];
    intro.classList.add('hidden'); gameover.classList.remove('show');
    centerPlayer(); updateHud(); spawnInitial(); toastMsg('Recover the packets');
  }

  function spawnInitial(){
    for(let i=0;i<6;i++) spawnPacket();
    for(let i=0;i<3;i++) spawnGlitch();
  }

  function randEdge(){
    const side = Math.floor(Math.random()*4);
    if(side===0) return {x:Math.random()*innerWidth,y:-20};
    if(side===1) return {x:innerWidth+20,y:Math.random()*innerHeight};
    if(side===2) return {x:Math.random()*innerWidth,y:innerHeight+20};
    return {x:-20,y:Math.random()*innerHeight};
  }

  function spawnPacket(){
    const m = 40;
    state.packets.push({x:m+Math.random()*(innerWidth-m*2), y:m+Math.random()*(innerHeight-m*2), r:10, pulse:Math.random()*Math.PI*2});
  }

  function spawnGlitch(){
    const p = randEdge();
    const angle = Math.atan2(player.y - p.y + (Math.random()-.5)*180, player.x - p.x + (Math.random()-.5)*180);
    const speed = 80 + Math.random()*80 + state.score*4;
    state.glitches.push({x:p.x,y:p.y,r:12 + Math.random()*8,vx:Math.cos(angle)*speed,vy:Math.sin(angle)*speed,spin:Math.random()*6-3});
  }

  function burst(x,y,color,count=14){
    for(let i=0;i<count;i++){
      const a = Math.random()*Math.PI*2, s = Math.random()*180 + 30;
      state.particles.push({x,y,vx:Math.cos(a)*s,vy:Math.sin(a)*s,life:.7+Math.random()*.4,max:1,color});
    }
  }

  function toastMsg(text){
    toast.textContent = text; toast.classList.add('show');
    clearTimeout(toast._t); toast._t = setTimeout(()=>toast.classList.remove('show'), 1200);
  }

  function updateHud(){
    scoreEl.textContent = state.score;
    healthEl.textContent = state.health;
    statusEl.textContent = state.won ? '200: Route restored' : (state.running ? 'Searching for route…' : '404: Page not found');
  }

  function dist(a,b){ const dx=a.x-b.x, dy=a.y-b.y; return Math.hypot(dx,dy); }

  function input(dt){
    let ax = 0, ay = 0;
    if(state.keys.has('arrowleft')||state.keys.has('a')) ax -= 1;
    if(state.keys.has('arrowright')||state.keys.has('d')) ax += 1;
    if(state.keys.has('arrowup')||state.keys.has('w')) ay -= 1;
    if(state.keys.has('arrowdown')||state.keys.has('s')) ay += 1;
    const len = Math.hypot(ax,ay)||1;
    const speed = player.dash>0 ? 520 : player.speed;
    player.vx = (ax/len)*speed; player.vy = (ay/len)*speed;
    player.x += player.vx * dt; player.y += player.vy * dt;
    player.x = Math.max(player.r, Math.min(innerWidth-player.r, player.x));
    player.y = Math.max(player.r, Math.min(innerHeight-player.r, player.y));
    if(state.dashCd>0) state.dashCd -= dt;
    if(player.dash>0) player.dash -= dt;
    player.trail.push({x:player.x,y:player.y,life:.35});
    if(player.trail.length>12) player.trail.shift();
  }

  function doDash(){
    if(!state.running || state.dashCd>0) return;
    player.dash = .14; state.dashCd = .8; burst(player.x, player.y, '97,232,238', 12); toastMsg('Dash');
  }

  function update(dt){
    if(!state.running) return;
    state.time += dt; state.shake = Math.max(0, state.shake - dt*3);
    input(dt);
    if(Math.random() < 0.015 + state.score*0.0008) spawnPacket();
    state.spawnTimer += dt;
    if(state.spawnTimer > Math.max(.45, 1.2 - state.score*0.03)){ state.spawnTimer = 0; spawnGlitch(); }

    state.packets.forEach(p => p.pulse += dt*4);
    state.packets = state.packets.filter(p => {
      if(dist(player,p) < player.r + p.r + 2){
        state.score += 1; updateHud(); burst(p.x,p.y,'97,232,238',18); toastMsg('Packet recovered');
        if(state.score >= state.target){ win(); return false; }
        return false;
      }
      return true;
    });

    state.glitches.forEach(g => {
      const angle = Math.atan2(player.y-g.y, player.x-g.x);
      const accel = 18 + Math.min(30, state.score*1.8);
      g.vx += Math.cos(angle)*accel*dt; g.vy += Math.sin(angle)*accel*dt;
      const cap = 170 + state.score*7;
      const sp = Math.hypot(g.vx,g.vy) || 1;
      if(sp > cap){ g.vx = g.vx/sp*cap; g.vy = g.vy/sp*cap; }
      g.x += g.vx*dt; g.y += g.vy*dt;
      if(dist(player,g) < player.r + g.r){
        if(player.dash>0){ burst(g.x,g.y,'255,106,130',16); g.dead = true; state.score += 1; updateHud(); toastMsg('Glitch erased'); }
        else { g.dead = true; hit(); }
      }
    });
    state.glitches = state.glitches.filter(g => !g.dead && g.x>-80 && g.x<innerWidth+80 && g.y>-80 && g.y<innerHeight+80);

    state.particles.forEach(p => { p.life -= dt; p.x += p.vx*dt; p.y += p.vy*dt; p.vx *= .98; p.vy *= .98; });
    state.particles = state.particles.filter(p => p.life > 0);
    player.trail.forEach(t => t.life -= dt); player.trail = player.trail.filter(t => t.life > 0);
  }

  function hit(){
    state.health -= 1; state.shake = .8; updateHud(); burst(player.x, player.y, '255,106,130', 22); toastMsg('Signal damaged');
    if(state.health <= 0) lose();
  }

  function lose(){
    state.running = false; state.best = Math.max(state.best, state.score); finalScore.textContent = state.score; bestScore.textContent = state.best;
    overKicker.textContent = 'Transmission ended'; overTitle.textContent = 'Signal lost.'; overText.textContent = 'The void swallowed the route before all packets were recovered.';
    gameover.classList.add('show'); updateHud();
  }

  function win(){
    state.running = false; state.won = true; state.best = Math.max(state.best, state.score); finalScore.textContent = state.score; bestScore.textContent = state.best;
    overKicker.textContent = 'Route rebuilt'; overTitle.textContent = 'Page found.'; overText.textContent = 'You stitched the missing packets back together and reopened the path home.';
    gameover.classList.add('show'); burst(player.x, player.y, '97,232,238', 40); updateHud();
  }

  function drawGrid(){
    ctx.save();
    const offsetX = (state.time*20)%40, offsetY = (state.time*12)%40;
    ctx.strokeStyle = 'rgba(125,160,255,0.08)'; ctx.lineWidth = 1;
    for(let x=-40+offsetX;x<innerWidth+40;x+=40){ ctx.beginPath(); ctx.moveTo(x,0); ctx.lineTo(x,innerHeight); ctx.stroke(); }
    for(let y=-40+offsetY;y<innerHeight+40;y+=40){ ctx.beginPath(); ctx.moveTo(0,y); ctx.lineTo(innerWidth,y); ctx.stroke(); }
    ctx.restore();
  }

  function draw(){
    ctx.clearRect(0,0,innerWidth,innerHeight);
    const shakeX = state.shake ? (Math.random()-.5)*10*state.shake : 0;
    const shakeY = state.shake ? (Math.random()-.5)*10*state.shake : 0;
    ctx.save(); ctx.translate(shakeX, shakeY);

    for(const s of state.stars){
      const tw = .35 + .65*Math.sin(state.time*s.z + s.x*.01);
      ctx.fillStyle = `rgba(255,255,255,${0.12 + tw*0.4})`;
      ctx.fillRect((s.x + state.time*s.z*8)%innerWidth, s.y, s.s, s.s);
    }
    drawGrid();

    for(const t of player.trail){
      ctx.beginPath(); ctx.fillStyle = `rgba(97,232,238,${t.life*.25})`; ctx.arc(t.x,t.y,player.r*(.4+t.life),0,Math.PI*2); ctx.fill();
    }

    for(const p of state.packets){
      const rr = p.r + Math.sin(p.pulse)*2;
      ctx.beginPath(); ctx.fillStyle = 'rgba(97,232,238,.12)'; ctx.arc(p.x,p.y,rr+8,0,Math.PI*2); ctx.fill();
      ctx.beginPath(); ctx.fillStyle = '#61e8ee'; ctx.arc(p.x,p.y,rr,0,Math.PI*2); ctx.fill();
      ctx.fillStyle = '#031719'; ctx.fillRect(p.x-3,p.y-6,6,12); ctx.fillRect(p.x-6,p.y-3,12,6);
    }

    for(const g of state.glitches){
      ctx.save(); ctx.translate(g.x,g.y); ctx.rotate(state.time*g.spin);
      ctx.fillStyle = 'rgba(255,106,130,.12)'; ctx.fillRect(-g.r-8,-g.r-8,(g.r+8)*2,(g.r+8)*2);
      ctx.strokeStyle = '#ff6a82'; ctx.lineWidth = 2;
      ctx.strokeRect(-g.r,-g.r,g.r*2,g.r*2);
      ctx.beginPath(); ctx.moveTo(-g.r, -g.r*.3); ctx.lineTo(g.r, g.r*.4); ctx.moveTo(-g.r*.5,g.r); ctx.lineTo(g.r*.5,-g.r); ctx.stroke();
      ctx.restore();
    }

    const look = Math.atan2(pointer.y-player.y, pointer.x-player.x);
    ctx.save(); ctx.translate(player.x,player.y); ctx.rotate(look);
    ctx.beginPath(); ctx.fillStyle = player.dash>0 ? '#ffffff' : '#2da7ff'; ctx.arc(0,0,player.r+4,0,Math.PI*2); ctx.fill();
    ctx.beginPath(); ctx.fillStyle = '#61e8ee'; ctx.arc(0,0,player.r,0,Math.PI*2); ctx.fill();
    ctx.fillStyle = '#031719'; ctx.fillRect(2,-4,10,8);
    ctx.beginPath(); ctx.arc(-3,-4,2,0,Math.PI*2); ctx.arc(-3,4,2,0,Math.PI*2); ctx.fill();
    ctx.restore();

    for(const p of state.particles){
      ctx.fillStyle = `rgba(${p.color},${Math.max(0,p.life/p.max)})`;
      ctx.fillRect(p.x,p.y,3,3);
    }

    ctx.restore();
  }

  let last = 0;
  function loop(ts){
    const dt = Math.min(.033, (ts-last)/1000 || .016); last = ts;
    update(dt); draw(); requestAnimationFrame(loop);
  }

  addEventListener('keydown', e => {
    const k = e.key.toLowerCase();
    if(['arrowup','arrowdown','arrowleft','arrowright',' ','a','s','d','w','r'].includes(k)) e.preventDefault();
    if(k === ' ') doDash();
    if(k === 'r') reset();
    state.keys.add(k);
  });
  addEventListener('keyup', e => state.keys.delete(e.key.toLowerCase()));
  addEventListener('pointermove', e => { pointer.x = e.clientX; pointer.y = e.clientY; });
  document.getElementById('startBtn').addEventListener('click', reset);
  document.getElementById('restartBtn').addEventListener('click', reset);
  themeBtn.addEventListener('click', () => {
    const root = document.documentElement;
    const next = root.getAttribute('data-theme') === 'dark' ? 'light' : 'dark';
    root.setAttribute('data-theme', next);
    themeBtn.setAttribute('aria-label', `Switch to ${next === 'dark' ? 'light' : 'dark'} mode`);
  });

  resize(); centerPlayer(); buildStars(); updateHud(); requestAnimationFrame(loop); addEventListener('resize', resize);
})();
</script>
